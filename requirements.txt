# MF Facts — RAG FAQ assistant for HDFC Mutual Fund schemes
# Install:  pip install -r requirements.txt
#
# Deployment (Render): see render.yaml and the Deploy section of README.md.
# These are all Linux wheels, so the same file works for local and remote.
# No torch: embeddings go through ONNX Runtime via fastembed, so the install is
# a few hundred MB and a cold build is minutes rather than tens of minutes.

# --- RAG pipeline ---
chromadb>=0.5.0              # stage 5: persistent vector store
# Embeddings. fastembed runs all-MiniLM-L6-v2 through ONNX Runtime instead of
# torch: same model, same 384 dims, ~90 MB of weights instead of ~2 GB. That
# difference is what makes a free-tier Render instance viable. If you need
# torch for something else, sentence-transformers>=3.0.0 is the drop-in swap,
# but src/embed.py must be changed back to SentenceTransformer.
fastembed>=0.8.0

# --- Ingestion (stages 1-2) ---
requests>=2.32.0             # fetch the source pages
beautifulsoup4>=4.12.0       # HTML -> text
lxml>=5.0.0                  # bs4 parser; the 450 KB pages are malformed
                             # Next.js output, and html.parser is materially
                             # worse on them

# --- Generation (stage 9) ---
# litellm covers OpenAI, Gemini and Ollama behind one API, so the swappable
# backend in src/generate.py needs a single client.
litellm>=1.40.0

# --- App ---
streamlit>=1.38.0            # UI
python-dotenv>=1.0.0         # loads .env (API keys)
